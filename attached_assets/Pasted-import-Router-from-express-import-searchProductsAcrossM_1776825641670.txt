import { Router } from "express"
import { searchProductsAcrossMerchants } from "../services/productSearchService"
import { getProductsForWardrobeGap } from "../services/wardrobeGapCommerceService"
import type { Merchant, WardrobeGap } from "../types/shop"

const router = Router()

router.post("/products/search", async (req, res) => {
  try {
    const {
      query,
      merchants,
      category,
      colorPreferences,
      styleTags,
      matchedWardrobeGapId,
      matchedWardrobeGapTitle,
      limitPerMerchant,
    } = req.body as {
      query: string
      merchants?: Merchant[]
      category?: string
      colorPreferences?: string[]
      styleTags?: string[]
      matchedWardrobeGapId?: string
      matchedWardrobeGapTitle?: string
      limitPerMerchant?: number
    }

    if (!query || typeof query !== "string") {
      return res.status(400).json({ error: "query is required" })
    }

    const products = await searchProductsAcrossMerchants({
      query,
      merchants,
      category,
      colorPreferences,
      styleTags,
      matchedWardrobeGapId,
      matchedWardrobeGapTitle,
      limitPerMerchant,
    })

    return res.json({ products })
  } catch (error) {
    console.error("POST /products/search failed", error)
    return res.status(500).json({ error: "failed_to_search_products" })
  }
})

router.post("/products/by-gap", async (req, res) => {
  try {
    const {
      gap,
      merchants,
      limitPerMerchant,
    } = req.body as {
      gap: WardrobeGap
      merchants?: Merchant[]
      limitPerMerchant?: number
    }

    if (!gap?.searchQuery) {
      return res.status(400).json({ error: "valid gap is required" })
    }

    const products = await getProductsForWardrobeGap({
      gap,
      merchants,
      limitPerMerchant,
    })

    return res.json({ products })
  } catch (error) {
    console.error("POST /products/by-gap failed", error)
    return res.status(500).json({ error: "failed_to_search_products_by_gap" })
  }
})

export default router
