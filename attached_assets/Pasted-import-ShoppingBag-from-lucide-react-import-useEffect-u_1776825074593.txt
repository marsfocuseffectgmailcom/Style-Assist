import { ShoppingBag } from "lucide-react"
import { useEffect, useMemo, useState } from "react"
import { AppShell } from "../components/AppShell"
import { PrimaryButton } from "../components/PrimaryButton"
import { Card } from "../components/Card"
import { SectionHeader } from "../components/SectionHeader"
import { ProductCard } from "../components/ProductCard"
import { WardrobeGapCard } from "../components/WardrobeGapCard"
import { fetchShoppingRecommendations, fetchRecommendedProducts } from "../lib/shopService"
import type { Merchant, RecommendedProduct, WardrobeGap } from "../types/shop"
import { CategoryPill } from "../components/CategoryPill"

const merchantFilters: Array<Merchant | "All"> = ["All", "Amazon", "SHEIN", "Temu", "ASOS"]

export default function Shop() {
  const [loading, setLoading] = useState(true)
  const [merchant, setMerchant] = useState<Merchant | "All">("All")
  const [wardrobeGaps, setWardrobeGaps] = useState<WardrobeGap[]>([])
  const [products, setProducts] = useState<RecommendedProduct[]>([])
  const [selectedGapId, setSelectedGapId] = useState<string | null>(null)

  useEffect(() => {
    async function load() {
      setLoading(true)
      const data = await fetchShoppingRecommendations()
      setWardrobeGaps(data.wardrobeGaps)
      setProducts(data.recommendedProducts)
      setLoading(false)
    }

    load()
  }, [])

  useEffect(() => {
    async function reloadProducts() {
      const nextProducts = await fetchRecommendedProducts({
        merchant,
      })
      setProducts(nextProducts)
    }

    reloadProducts()
  }, [merchant])

  const visibleProducts = useMemo(() => {
    if (!selectedGapId) return products
    return products.filter((product) => product.matchedWardrobeGapId === selectedGapId)
  }, [products, selectedGapId])

  return (
    <AppShell>
      <header className="mb-5 pt-4">
        <div className="flex items-center gap-3">
          <div className="flex h-11 w-11 items-center justify-center rounded-full bg-white/5">
            <ShoppingBag size={20} className="text-[#C8A96A]" />
          </div>
          <div>
            <h1 className="text-[28px] font-semibold tracking-[-0.3px]">Shop</h1>
            <p className="text-sm text-[#A8AFBE]">
              Monetisable wardrobe upgrades and affiliate-ready recommendations
            </p>
          </div>
        </div>
      </header>

      <Card
        elevated
        className="bg-gradient-to-br from-[#3A2922] via-[#241A18] to-[#151922]"
      >
        <h2 className="text-[18px] font-semibold">Wardrobe Gap Analysis</h2>
        <p className="mt-2 text-sm leading-6 text-[#D6D0C8]">
          The smartest purchases are the ones that unlock more outfits — not more clutter
        </p>

        <div className="mt-4">
          <PrimaryButton fullWidth={false} className="px-4">
            Start 7-Day Free Trial
          </PrimaryButton>
        </div>
      </Card>

      <section className="mt-6">
        <SectionHeader title="Retailers" />
        <div className="flex gap-2 overflow-x-auto pb-1">
          {merchantFilters.map((filter) => (
            <CategoryPill
              key={filter}
              active={merchant === filter}
              onClick={() => setMerchant(filter)}
            >
              {filter}
            </CategoryPill>
          ))}
        </div>
      </section>

      <section className="mt-6">
        <SectionHeader title="Wardrobe Gaps" />
        <div className="space-y-3">
          {loading ? (
            <Card>Loading wardrobe gaps...</Card>
          ) : (
            wardrobeGaps.map((gap) => (
              <WardrobeGapCard
                key={gap.id}
                gap={gap}
                onShop={() => setSelectedGapId(gap.id)}
              />
            ))
          )}
        </div>
      </section>

      <section className="mt-6">
        <SectionHeader
          title={selectedGapId ? "Products For Selected Gap" : "Recommended Products"}
          actionLabel={selectedGapId ? "Clear filter" : undefined}
          onActionClick={selectedGapId ? () => setSelectedGapId(null) : undefined}
        />

        <div className="space-y-3">
          {visibleProducts.map((product) => (
            <ProductCard
              key={product.id}
              product={product}
              sourceScreen="gap-analysis"
            />
          ))}
        </div>
      </section>
    </AppShell>
  )
}
