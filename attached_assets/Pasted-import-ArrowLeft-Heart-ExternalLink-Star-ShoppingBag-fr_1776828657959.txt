import { ArrowLeft, Heart, ExternalLink, Star, ShoppingBag } from "lucide-react"
import { Link, useLocation, useNavigate } from "react-router-dom"
import { AppShell } from "../components/AppShell"
import { Card } from "../components/Card"
import { PrimaryButton } from "../components/PrimaryButton"
import { useSavedProducts } from "../hooks/useSavedProducts"
import { useRecentlyViewed } from "../hooks/useRecentlyViewed"
import { useAffiliateClickTracking } from "../hooks/useAffiliateClickTracking"
import type { RecommendedProduct } from "../types/shop"
import { getProductRecommendationReason } from "../lib/productDetailInsights"

type ProductDetailsLocationState = {
  product?: RecommendedProduct
  sourceScreen?: "shop" | "stylist" | "gap-analysis" | "home"
}

export default function ProductDetails() {
  const navigate = useNavigate()
  const location = useLocation()
  const { product, sourceScreen } = (location.state || {}) as ProductDetailsLocationState

  const { isSaved, toggleSavedProduct } = useSavedProducts()
  const { addRecentlyViewed } = useRecentlyViewed()
  const { trackClick } = useAffiliateClickTracking({
    sourceScreen: sourceScreen || "shop",
  })

  if (!product) {
    return (
      <AppShell>
        <div className="pt-6">
          <Card className="text-center">
            <ShoppingBag size={24} className="mx-auto text-[#C8A96A]" />
            <h1 className="mt-4 text-[22px] font-semibold text-[#F6F3EE]">
              Product not found
            </h1>
            <p className="mt-2 text-sm leading-6 text-[#A8AFBE]">
              This product could not be loaded. Go back to the shop and try again
            </p>

            <div className="mt-5">
              <Link to="/shop">
                <PrimaryButton fullWidth={false} className="px-5">
                  Back to Shop
                </PrimaryButton>
              </Link>
            </div>
          </Card>
        </div>
      </AppShell>
    )
  }

  const saved = isSaved(product.id)
  const reason = getProductRecommendationReason(product)

  function handleSave() {
    toggleSavedProduct(product)
  }

  function handleOpenAffiliate() {
    addRecentlyViewed(product)
    trackClick(product)
  }

  return (
    <AppShell>
      <header className="mb-4 flex items-center justify-between pt-4">
        <button
          type="button"
          onClick={() => navigate(-1)}
          className="flex h-10 w-10 items-center justify-center rounded-full bg-white/5 text-[#F6F3EE]"
        >
          <ArrowLeft size={18} />
        </button>

        <h1 className="text-[18px] font-semibold">Product Details</h1>

        <button
          type="button"
          onClick={handleSave}
          className={`flex h-10 w-10 items-center justify-center rounded-full border transition ${
            saved
              ? "border-[#FF4D8D]/30 bg-[#FF4D8D]/15 text-[#FF78A8]"
              : "border-white/10 bg-white/5 text-[#A8AFBE]"
          }`}
          aria-label={saved ? "Remove from saved products" : "Save product"}
        >
          <Heart size={18} className={saved ? "fill-current" : ""} />
        </button>
      </header>

      <section className="overflow-hidden rounded-[28px] border border-white/10 bg-[#151922]">
        <div className="relative h-[340px] overflow-hidden bg-[#11151C]">
          <img
            src={product.image}
            alt={product.name}
            className="h-full w-full object-cover"
          />

          <div className="absolute left-4 top-4 rounded-full bg-black/60 px-3 py-1 text-xs font-semibold text-white backdrop-blur-sm">
            {product.merchant}
          </div>

          {product.commissionEligible ? (
            <div className="absolute right-4 top-4 rounded-full bg-[#1C2A20]/85 px-3 py-1 text-xs font-semibold text-[#8FD3C1] backdrop-blur-sm">
              Affiliate
            </div>
          ) : null}
        </div>

        <div className="p-4">
          <div className="flex items-start justify-between gap-4">
            <div className="min-w-0">
              <h2 className="text-[22px] font-semibold leading-tight text-[#F6F3EE]">
                {product.name}
              </h2>
              <p className="mt-1 text-sm uppercase tracking-[0.12em] text-[#6F7788]">
                {product.brand}
              </p>
            </div>

            <div className="shrink-0 text-right">
              <p className="text-[18px] font-semibold text-[#F6F3EE]">
                {product.currency} {product.price}
              </p>
            </div>
          </div>

          <div className="mt-4 flex flex-wrap gap-2">
            {product.rating ? (
              <div className="inline-flex items-center gap-1 rounded-full bg-white/5 px-3 py-1 text-xs text-[#F6F3EE]">
                <Star size={12} className="text-[#C8A96A]" />
                {product.rating.toFixed(1)}
                {typeof product.reviewCount === "number" ? ` (${product.reviewCount})` : ""}
              </div>
            ) : null}

            {typeof product.inStock === "boolean" ? (
              <div
                className={`inline-flex items-center rounded-full px-3 py-1 text-xs ${
                  product.inStock
                    ? "bg-[#1C2A20] text-[#8FD3C1]"
                    : "bg-[#2A1C1C] text-[#F0A3A3]"
                }`}
              >
                {product.inStock ? "In stock" : "Out of stock"}
              </div>
            ) : null}

            {product.category ? (
              <div className="inline-flex items-center rounded-full bg-white/5 px-3 py-1 text-xs text-[#A8AFBE]">
                {product.category}
              </div>
            ) : null}
          </div>
        </div>
      </section>

      <Card className="mt-5">
        <h3 className="text-[16px] font-semibold">Why this was recommended</h3>
        <p className="mt-2 text-sm leading-6 text-[#A8AFBE]">{reason}</p>

        {product.matchedWardrobeGapTitle ? (
          <div className="mt-3 rounded-[18px] bg-white/5 p-3">
            <p className="text-xs uppercase tracking-[0.12em] text-[#6F7788]">
              Wardrobe gap match
            </p>
            <p className="mt-1 text-sm font-medium text-[#F6F3EE]">
              {product.matchedWardrobeGapTitle}
            </p>
          </div>
        ) : null}
      </Card>

      {(product.colors?.length || product.availableSizes?.length || product.styleTags?.length) ? (
        <Card className="mt-5">
          <h3 className="text-[16px] font-semibold">Details</h3>

          {product.colors?.length ? (
            <div className="mt-3">
              <p className="text-xs uppercase tracking-[0.12em] text-[#6F7788]">
                Colours
              </p>
              <div className="mt-2 flex flex-wrap gap-2">
                {product.colors.map((color) => (
                  <span
                    key={color}
                    className="rounded-full border border-white/10 bg-white/5 px-3 py-1 text-xs text-[#F6F3EE]"
                  >
                    {color}
                  </span>
                ))}
              </div>
            </div>
          ) : null}

          {product.availableSizes?.length ? (
            <div className="mt-4">
              <p className="text-xs uppercase tracking-[0.12em] text-[#6F7788]">
                Sizes
              </p>
              <div className="mt-2 flex flex-wrap gap-2">
                {product.availableSizes.map((size) => (
                  <span
                    key={size}
                    className="rounded-full border border-white/10 bg-white/5 px-3 py-1 text-xs text-[#F6F3EE]"
                  >
                    {size}
                  </span>
                ))}
              </div>
            </div>
          ) : null}

          {product.styleTags?.length ? (
            <div className="mt-4">
              <p className="text-xs uppercase tracking-[0.12em] text-[#6F7788]">
                Style tags
              </p>
              <div className="mt-2 flex flex-wrap gap-2">
                {product.styleTags.map((tag) => (
                  <span
                    key={tag}
                    className="rounded-full border border-white/10 bg-white/5 px-3 py-1 text-xs text-[#C8A96A]"
                  >
                    {tag}
                  </span>
                ))}
              </div>
            </div>
          ) : null}
        </Card>
      ) : null}

      <div className="mt-6 space-y-3">
        <PrimaryButton onClick={handleOpenAffiliate}>
          <span className="inline-flex items-center gap-2">
            View at {product.merchant}
            <ExternalLink size={16} />
          </span>
        </PrimaryButton>

        <button
          type="button"
          onClick={handleSave}
          className="h-11 w-full rounded-[18px] border border-white/10 bg-white/5 text-sm text-[#F6F3EE] transition active:scale-[0.98]"
        >
          {saved ? "Remove from Saved" : "Save for Later"}
        </button>
      </div>
    </AppShell>
  )
}
