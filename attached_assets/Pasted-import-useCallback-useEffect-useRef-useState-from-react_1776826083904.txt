import { useCallback, useEffect, useRef, useState } from "react"
import type { Merchant, RecommendedProduct } from "../types/shop"
import { searchProducts } from "../lib/api/shopApi"

type UseProductSearchArgs = {
  initialQuery?: string
  merchants?: Merchant[]
  debounceMs?: number
  autoSearch?: boolean
}

type UseProductSearchResult = {
  query: string
  setQuery: (q: string) => void
  products: RecommendedProduct[]
  loading: boolean
  error: string | null
  searchNow: (q?: string) => Promise<void>
  clear: () => void
}

export function useProductSearch({
  initialQuery = "",
  merchants,
  debounceMs = 400,
  autoSearch = true,
}: UseProductSearchArgs = {}): UseProductSearchResult {
  const [query, setQueryState] = useState(initialQuery)
  const [products, setProducts] = useState<RecommendedProduct[]>([])
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const debounceRef = useRef<number | null>(null)
  const latestQueryRef = useRef(query)

  const runSearch = useCallback(
    async (q: string) => {
      if (!q || q.trim().length < 2) {
        setProducts([])
        return
      }

      try {
        setLoading(true)
        setError(null)

        const result = await searchProducts({
          query: q,
          merchants,
          limitPerMerchant: 6,
        })

        // prevent race condition overwrite
        if (latestQueryRef.current === q) {
          setProducts(result)
        }
      } catch (err) {
        console.error("useProductSearch failed", err)
        setError(err instanceof Error ? err.message : "Search failed")
      } finally {
        setLoading(false)
      }
    },
    [merchants],
  )

  const setQuery = useCallback(
    (q: string) => {
      latestQueryRef.current = q
      setQueryState(q)

      if (!autoSearch) return

      if (debounceRef.current) {
        clearTimeout(debounceRef.current)
      }

      debounceRef.current = window.setTimeout(() => {
        runSearch(q)
      }, debounceMs)
    },
    [autoSearch, debounceMs, runSearch],
  )

  const searchNow = useCallback(
    async (q?: string) => {
      const finalQuery = q ?? query
      latestQueryRef.current = finalQuery
      await runSearch(finalQuery)
    },
    [query, runSearch],
  )

  const clear = useCallback(() => {
    setQueryState("")
    setProducts([])
    setError(null)
    setLoading(false)
  }, [])

  useEffect(() => {
    if (initialQuery && autoSearch) {
      runSearch(initialQuery)
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  return {
    query,
    setQuery,
    products,
    loading,
    error,
    searchNow,
    clear,
  }
}
