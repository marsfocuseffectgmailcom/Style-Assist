import { Activity, Heart, Clock3, Trash2 } from "lucide-react"
import { Link } from "react-router-dom"
import { AppShell } from "../components/AppShell"
import { Card } from "../components/Card"
import { SectionHeader } from "../components/SectionHeader"
import { ProductCard } from "../components/ProductCard"
import { PrimaryButton } from "../components/PrimaryButton"
import { useSavedProducts } from "../hooks/useSavedProducts"
import { useRecentlyViewed } from "../hooks/useRecentlyViewed"

export default function MyActivity() {
  const {
    savedProducts,
    savedCount,
    clearSavedProducts,
  } = useSavedProducts()

  const {
    recentlyViewed,
    recentlyViewedCount,
    clearRecentlyViewed,
  } = useRecentlyViewed()

  const hasSavedProducts = savedProducts.length > 0
  const hasRecentlyViewed = recentlyViewed.length > 0
  const hasAnyActivity = hasSavedProducts || hasRecentlyViewed

  return (
    <AppShell>
      <header className="mb-5 pt-4">
        <div className="flex items-center gap-3">
          <div className="flex h-11 w-11 items-center justify-center rounded-full bg-white/5">
            <Activity size={20} className="text-[#C8A96A]" />
          </div>

          <div>
            <h1 className="text-[28px] font-semibold tracking-[-0.3px]">
              My Activity
            </h1>
            <p className="text-sm text-[#A8AFBE]">
              Saved products and your recent browsing history
            </p>
          </div>
        </div>
      </header>

      <Card
        elevated
        className="bg-gradient-to-br from-[#1D2230] via-[#171C27] to-[#151922]"
      >
        <div className="flex items-start justify-between gap-4">
          <div>
            <h2 className="text-[18px] font-semibold">Your Shopping Activity</h2>
            <p className="mt-2 text-sm leading-6 text-[#D6D0C8]">
              Keep track of what you saved and what you explored so your style journey stays connected
            </p>
          </div>

          <div className="flex flex-col gap-2">
            <div className="rounded-full bg-white/10 px-3 py-1 text-xs font-semibold text-[#F6F3EE]">
              {savedCount} saved
            </div>
            <div className="rounded-full bg-white/10 px-3 py-1 text-xs font-semibold text-[#F6F3EE]">
              {recentlyViewedCount} viewed
            </div>
          </div>
        </div>
      </Card>

      <section className="mt-6">
        <SectionHeader title="Saved Products" />

        {hasSavedProducts ? (
          <>
            <div className="mb-3">
              <button
                onClick={clearSavedProducts}
                className="inline-flex items-center gap-2 rounded-full border border-white/10 bg-white/5 px-4 py-2 text-sm text-[#F6F3EE] transition hover:bg-white/10"
              >
                <Trash2 size={15} />
                Clear saved
              </button>
            </div>

            <div className="space-y-3">
              {savedProducts.map((product) => (
                <ProductCard
                  key={product.id}
                  product={product}
                  sourceScreen="shop"
                  showMerchantBadge
                />
              ))}
            </div>
          </>
        ) : (
          <Card className="text-center">
            <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-white/5">
              <Heart size={22} className="text-[#FF78A8]" />
            </div>

            <h3 className="mt-4 text-[18px] font-semibold text-[#F6F3EE]">
              No saved products yet
            </h3>

            <p className="mx-auto mt-2 max-w-[280px] text-sm leading-6 text-[#A8AFBE]">
              Tap the heart on product cards to build a wishlist you can return to later
            </p>
          </Card>
        )}
      </section>

      <section className="mt-6">
        <SectionHeader title="Recently Viewed" />

        {hasRecentlyViewed ? (
          <>
            <div className="mb-3">
              <button
                onClick={clearRecentlyViewed}
                className="inline-flex items-center gap-2 rounded-full border border-white/10 bg-white/5 px-4 py-2 text-sm text-[#F6F3EE] transition hover:bg-white/10"
              >
                <Trash2 size={15} />
                Clear history
              </button>
            </div>

            <div className="space-y-3">
              {recentlyViewed.map((product) => (
                <ProductCard
                  key={product.id}
                  product={product}
                  sourceScreen="shop"
                  showMerchantBadge
                />
              ))}
            </div>
          </>
        ) : (
          <Card className="text-center">
            <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-white/5">
              <Clock3 size={22} className="text-[#C8A96A]" />
            </div>

            <h3 className="mt-4 text-[18px] font-semibold text-[#F6F3EE]">
              No recent views yet
            </h3>

            <p className="mx-auto mt-2 max-w-[280px] text-sm leading-6 text-[#A8AFBE]">
              Products you open from the shop will appear here so you can compare them later
            </p>
          </Card>
        )}
      </section>

      {!hasAnyActivity ? (
        <section className="mt-6">
          <Card className="text-center">
            <h3 className="text-[18px] font-semibold text-[#F6F3EE]">
              Start exploring
            </h3>

            <p className="mx-auto mt-2 max-w-[280px] text-sm leading-6 text-[#A8AFBE]">
              Browse product recommendations and start saving pieces that match your wardrobe gaps
            </p>

            <div className="mt-5">
              <Link to="/shop">
                <PrimaryButton fullWidth={false} className="px-5">
                  Go to Shop
                </PrimaryButton>
              </Link>
            </div>
          </Card>
        </section>
      ) : null}
    </AppShell>
  )
}
